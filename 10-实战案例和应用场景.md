#### 1.1.3 核心功能实现

**高并发订单处理**：
```sql
-- 创建订单的存储过程
DELIMITER //
CREATE PROCEDURE create_order(
    IN p_user_id BIGINT,
    IN p_items JSON,
    IN p_payment_method VARCHAR(20)
)
BEGIN
    DECLARE v_total_amount DECIMAL(12,2) DEFAULT 0;
    DECLARE v_order_no VARCHAR(32);
    DECLARE v_order_id BIGINT;
    DECLARE v_item_count INT;
    DECLARE i INT DEFAULT 0;
    DECLARE v_product_id BIGINT;
    DECLARE v_quantity INT;
    DECLARE v_unit_price DECIMAL(10,2);
    DECLARE v_stock INT;

    -- 生成订单号
    SET v_order_no = CONCAT('ORD', DATE_FORMAT(NOW(), '%Y%m%d'), LPAD(FLOOR(RAND() * 1000000), 6, '0'));

    -- 开始事务
    START TRANSACTION;

    BEGIN
        DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
        BEGIN
            ROLLBACK;
            SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = '订单创建失败';
        END;

        -- 计算总金额并检查库存
        SET v_item_count = JSON_LENGTH(p_items);
        WHILE i < v_item_count DO
            SET v_product_id = JSON_UNQUOTE(JSON_EXTRACT(p_items, CONCAT('$[', i, '].product_id')));
            SET v_quantity = JSON_UNQUOTE(JSON_EXTRACT(p_items, CONCAT('$[', i, '].quantity')));

            -- 获取商品价格和库存
            SELECT price, stock INTO v_unit_price, v_stock
            FROM products WHERE id = v_product_id FOR UPDATE;

            -- 检查库存
            IF v_stock < v_quantity THEN
                SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = CONCAT('商品ID ', v_product_id, ' 库存不足');
            END IF;

            -- 更新库存
            UPDATE products SET stock = stock - v_quantity WHERE id = v_product_id;

            -- 计算金额
            SET v_total_amount = v_total_amount + (v_unit_price * v_quantity);
            SET i = i + 1;
        END WHILE;

        -- 创建订单
        INSERT INTO orders (order_no, user_id, total_amount, payment_method, status)
        VALUES (v_order_no, p_user_id, v_total_amount, p_payment_method, 0);

        SET v_order_id = LAST_INSERT_ID();

        -- 创建订单明细
        SET i = 0;
        WHILE i < v_item_count DO
            SET v_product_id = JSON_UNQUOTE(JSON_EXTRACT(p_items, CONCAT('$[', i, '].product_id')));
            SET v_quantity = JSON_UNQUOTE(JSON_EXTRACT(p_items, CONCAT('$[', i, '].quantity')));
            SELECT price INTO v_unit_price FROM products WHERE id = v_product_id;

            INSERT INTO order_items (order_id, product_id, quantity, unit_price, total_price)
            VALUES (v_order_id, v_product_id, v_quantity, v_unit_price, v_unit_price * v_quantity);

            SET i = i + 1;
        END WHILE;

        -- 记录销售分析数据
        INSERT INTO sales_analytics (order_id, product_id, category_id, user_id, sale_date, region, amount, quantity, profit)
        SELECT
            v_order_id,
            oi.product_id,
            p.category_id,
            p_user_id,
            CURDATE(),
            '未知',  -- 实际项目中根据用户地址获取
            oi.total_price,
            oi.quantity,
            oi.total_price - (oi.quantity * p.cost)
        FROM order_items oi
        JOIN products p ON oi.product_id = p.id
        WHERE oi.order_id = v_order_id;

        COMMIT;

        -- 返回订单信息
        SELECT v_order_id as order_id, v_order_no as order_no, v_total_amount as total_amount;
    END;
END //
DELIMITER ;
```

**实时库存管理**：
```sql
-- 库存变动监控视图
CREATE VIEW inventory_monitor AS
SELECT
    p.id,
    p.name,
    p.category_id,
    p.stock,
    p.price,
    COALESCE(sold.sold_quantity, 0) as sold_quantity,
    p.stock - COALESCE(sold.sold_quantity, 0) as available_stock,
    CASE
        WHEN p.stock - COALESCE(sold.sold_quantity, 0) <= 10 THEN '库存告急'
        WHEN p.stock - COALESCE(sold.sold_quantity, 0) <= 50 THEN '库存偏低'
        ELSE '库存正常'
    END as stock_status,
    p.updated_at
FROM products p
LEFT JOIN (
    SELECT
        product_id,
        SUM(quantity) as sold_quantity
    FROM order_items oi
    JOIN orders o ON oi.order_id = o.id
    WHERE o.status IN (1, 2, 3)  -- 已支付、已发货、已完成
    GROUP BY product_id
) sold ON p.id = sold.product_id;

-- 低库存告警查询
SELECT * FROM inventory_monitor
WHERE stock_status IN ('库存告急', '库存偏低')
ORDER BY available_stock;
```

**实时销售分析**：
```sql
-- 实时销售仪表板查询
SELECT
    DATE(sale_date) as sale_date,
    COUNT(DISTINCT order_id) as order_count,
    COUNT(DISTINCT user_id) as customer_count,
    SUM(amount) as total_sales,
    SUM(quantity) as total_quantity,
    SUM(profit) as total_profit,
    AVG(amount) as avg_order_value
FROM sales_analytics
WHERE sale_date >= DATE_SUB(CURDATE(), INTERVAL 7 DAY)
GROUP BY DATE(sale_date)
ORDER BY sale_date;

-- 商品销售排名（使用 TiFlash 加速）
SELECT /*+ TIFLASH() */
    p.name as product_name,
    p.category_id,
    COUNT(DISTINCT sa.order_id) as order_count,
    SUM(sa.quantity) as total_quantity,
    SUM(sa.amount) as total_amount,
    SUM(sa.profit) as total_profit,
    RANK() OVER (ORDER BY SUM(sa.amount) DESC) as sales_rank
FROM sales_analytics sa
JOIN products p ON sa.product_id = p.id
WHERE sa.sale_date >= DATE_SUB(CURDATE(), INTERVAL 30 DAY)
GROUP BY p.id, p.name, p.category_id
ORDER BY total_amount DESC
LIMIT 20;

-- 用户购买行为分析
SELECT /*+ TIFLASH() */
    u.id as user_id,
    u.username,
    COUNT(DISTINCT sa.order_id) as order_count,
    SUM(sa.amount) as total_spent,
    AVG(sa.amount) as avg_order_value,
    DATEDIFF(MAX(sa.sale_date), MIN(sa.sale_date)) as customer_lifetime_days,
    CASE
        WHEN COUNT(DISTINCT sa.order_id) >= 10 THEN 'VIP客户'
        WHEN COUNT(DISTINCT sa.order_id) >= 5 THEN '活跃客户'
        WHEN COUNT(DISTINCT sa.order_id) >= 1 THEN '新客户'
        ELSE '潜在客户'
    END as customer_level
FROM sales_analytics sa
JOIN users u ON sa.user_id = u.id
WHERE sa.sale_date >= DATE_SUB(CURDATE(), INTERVAL 90 DAY)
GROUP BY u.id, u.username
HAVING total_spent > 0
ORDER BY total_spent DESC
LIMIT 100;
```

**读写分离架构**：
```java
// Java 读写分离配置示例
@Configuration
public class DataSourceConfig {

    @Bean
    @ConfigurationProperties(prefix = "spring.datasource.master")
    public DataSource masterDataSource() {
        return DataSourceBuilder.create().build();
    }

    @Bean
    @ConfigurationProperties(prefix = "spring.datasource.slave")
    public DataSource slaveDataSource() {
        return DataSourceBuilder.create().build();
    }

    @Bean
    public DataSource routingDataSource() {
        DynamicDataSource routingDataSource = new DynamicDataSource();
        Map<Object, Object> dataSourceMap = new HashMap<>();
        dataSourceMap.put("master", masterDataSource());
        dataSourceMap.put("slave", slaveDataSource());
        routingDataSource.setTargetDataSources(dataSourceMap);
        routingDataSource.setDefaultTargetDataSource(masterDataSource());
        return routingDataSource;
    }
}

// 读写分离注解
@Target({ElementType.METHOD})
@Retention(RetentionPolicy.RUNTIME)
public @interface ReadOnly {
}

// AOP 切面实现读写分离
@Aspect
@Component
public class DataSourceAspect {

    @Autowired
    private DynamicDataSource dataSource;

    @Around("@annotation(readOnly)")
    public Object determineDataSource(ProceedingJoinPoint joinPoint, ReadOnly readOnly) throws Throwable {
        try {
            if (readOnly != null) {
                DataSourceContextHolder.setDataSourceKey("slave");
            }
            return joinPoint.proceed();
        } finally {
            DataSourceContextHolder.clearDataSourceKey();
        }
    }
}
```

```java
public class CacheService {

    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    // 缓存商品信息
    public void cacheProduct(Long productId, Product product) {
        String key = "product:" + productId;

    }

    // 获取缓存商品
    public Product getCachedProduct(Long productId) {
        String key = "product:" + productId;
        return (Product) redisTemplate.opsForValue().get(key);
    }

    // 缓存用户信息
    public void cacheUser(Long userId, User user) {
        String key = "user:" + userId;
        redisTemplate.opsForValue().set(key, user, 24, TimeUnit.HOURS);
    }

    // 删除缓存
    public void evictCache(String pattern) {
        Set<String> keys = redisTemplate.keys(pattern);
        if (keys != null) {
            redisTemplate.delete(keys);
        }
    }
```

**秒杀实现逻辑**：
```java
@Service
public class SeckillService {

    @Autowired
    private JdbcTemplate jdbcTemplate;

    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    @Autowired
    private CacheService cacheService;

    // 秒杀下单
    @Transactional
    public SeckillOrder seckillOrder(Long activityId, Long userId, int quantity) {
        // 1. 检查活动状态
        SeckillActivity activity = checkActivityStatus(activityId);

        // 2. 检查用户是否已参与
        checkUserParticipation(activityId, userId);

        // 3. 库存预扣减（Redis 原子操作）
        String stockKey = "seckill:stock:" + activityId;
        Long remaining = redisTemplate.opsForValue().decrement(stockKey, quantity);

        if (remaining < 0) {
            // 库存不足，恢复库存
            redisTemplate.opsForValue().increment(stockKey, quantity);
            throw new BusinessException("库存不足");
        }

        try {
            // 4. 创建秒杀订单
            SeckillOrder order = createSeckillOrder(activity, userId, quantity);

            // 5. 更新数据库库存
            updateActivityStock(activityId, quantity);

            // 6. 发送消息队列进行后续处理
            sendMessage("seckill_order_created", order);

            return order;

        } catch (Exception e) {
            // 异常情况恢复库存
            redisTemplate.opsForValue().increment(stockKey, quantity);
            throw new BusinessException("秒杀失败: " + e.getMessage());
        }
    }

    private SeckillActivity checkActivityStatus(Long activityId) {
        String cacheKey = "seckill:activity:" + activityId;
        SeckillActivity activity = (SeckillActivity) cacheService.get(cacheKey);

        if (activity == null) {
            activity = seckillActivityMapper.selectById(activityId);
            if (activity == null) {
                throw new BusinessException("活动不存在");
            }
            cacheService.set(cacheKey, activity, 5, TimeUnit.MINUTES);
        }

        if (activity.getStatus() != 1) {
            throw new BusinessException("活动未开始或已结束");
        }

        if (activity.getStockCount() <= 0) {
            throw new BusinessException("库存已售罄");
        }

        return activity;
    }
}
```

#### 2.1.3 核心业务实现

**转账交易实现**：
```sql
-- 转账存储过程
DELIMITER //
CREATE PROCEDURE transfer_funds(
    IN p_from_account VARCHAR(30),
    IN p_to_account VARCHAR(30),
    IN p_amount DECIMAL(18,2),
    IN p_currency VARCHAR(3),
    IN p_description VARCHAR(200),
    OUT p_transaction_number VARCHAR(32),
    OUT p_result_code INT,
    OUT p_result_message VARCHAR(200)
)
BEGIN
    DECLARE v_from_account_id BIGINT;
    DECLARE v_to_account_id BIGINT;
    DECLARE v_from_balance DECIMAL(18,2);
    DECLARE v_to_balance DECIMAL(18,2);
    DECLARE v_transaction_number VARCHAR(32);
    DECLARE v_fee DECIMAL(10,2) DEFAULT 0.00;
    DECLARE v_total_amount DECIMAL(18,2);

    -- 初始化返回值
    SET p_result_code = 0;
    SET p_result_message = '转账成功';

    -- 生成交易号
    SET v_transaction_number = CONCAT('TXN', DATE_FORMAT(NOW(), '%Y%m%d%H%i%s'), LPAD(FLOOR(RAND() * 1000), 3, '0'));
    SET p_transaction_number = v_transaction_number;

    -- 计算手续费（示例：转账金额的 0.1%，最低 1 元）
    SET v_fee = GREATEST(p_amount * 0.001, 1.00);
    SET v_total_amount = p_amount + v_fee;

    -- 开始事务
    START TRANSACTION;

    BEGIN
        DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
        BEGIN
            ROLLBACK;
            SET p_result_code = -1;
            SET p_result_message = '转账失败：系统错误';
        END;

        -- 检查转出账户
        SELECT id, balance INTO v_from_account_id, v_from_balance
        FROM accounts WHERE account_number = p_from_account AND status = 1 FOR UPDATE;

        IF v_from_account_id IS NULL THEN
            SET p_result_code = -2;
            SET p_result_message = '转出账户不存在或已冻结';
            LEAVE transfer_block;
        END IF;

        -- 检查转入账户
        SELECT id, balance INTO v_to_account_id, v_to_balance
        FROM accounts WHERE account_number = p_to_account AND status = 1 FOR UPDATE;

        IF v_to_account_id IS NULL THEN
            SET p_result_code = -3;
            SET p_result_message = '转入账户不存在或已冻结';
            LEAVE transfer_block;
        END IF;

        -- 检查余额
        IF v_from_balance < v_total_amount THEN
            SET p_result_code = -4;
            SET p_result_message = '余额不足';
            LEAVE transfer_block;
        END IF;

        -- 风险控制检查
        IF NOT check_risk_control(v_from_account_id, p_amount) THEN
            SET p_result_code = -5;
            SET p_result_message = '交易被风控系统拦截';
            LEAVE transfer_block;
        END IF;

        -- 更新账户余额
        UPDATE accounts SET balance = balance - v_total_amount WHERE id = v_from_account_id;
        UPDATE accounts SET balance = balance + p_amount WHERE id = v_to_account_id;

        -- 记录交易
        INSERT INTO transactions (
            transaction_number, from_account_id, to_account_id,
            amount, currency, transaction_type, status, description
        ) VALUES (
            v_transaction_number, v_from_account_id, v_to_account_id,
            p_amount, p_currency, 3, 1, p_description
        );

        -- 记录手续费交易
        IF v_fee > 0 THEN
            INSERT INTO transactions (
                transaction_number, from_account_id, amount,
                currency, transaction_type, status, description
            ) VALUES (
                CONCAT(v_transaction_number, '_FEE'), v_from_account_id, v_fee,
                p_currency, 4, 1, '转账手续费'
            );
        END IF;

        COMMIT;

        -- 发送通知
        CALL send_transaction_notification(v_from_account_id, v_transaction_number, '转账支出');
        CALL send_transaction_notification(v_to_account_id, v_transaction_number, '转账收入');

    END transfer_block;
END //

CREATE FUNCTION check_risk_control(p_account_id BIGINT, p_amount DECIMAL(18,2))
RETURNS BOOLEAN
DETERMINISTIC
BEGIN
    DECLARE v_risk_score INT DEFAULT 0;
    DECLARE v_daily_total DECIMAL(18,2) DEFAULT 0;
    DECLARE v_account_limit DECIMAL(18,2);

    -- 检查单笔交易限额
    SELECT daily_transfer_limit INTO v_account_limit
    FROM account_limits WHERE account_id = p_account_id;

    IF v_account_limit IS NOT NULL AND p_amount > v_account_limit THEN
        -- 记录风控日志
        INSERT INTO risk_control_logs (
            account_id, risk_type, risk_level, description, action_taken
        ) VALUES (
            p_account_id, '单笔超限', 2,
            CONCAT('交易金额 ', p_amount, ' 超过限额 ', v_account_limit),
            '拒绝交易'
        );
        RETURN FALSE;
    END IF;

    -- 检查当日累计交易
    SELECT COALESCE(SUM(amount), 0) INTO v_daily_total
    FROM transactions
    WHERE from_account_id = p_account_id
    AND DATE(created_at) = CURDATE()
    AND status = 1;

    IF v_daily_total + p_amount > 50000 THEN  -- 日累计限额 5万
        INSERT INTO risk_control_logs (
            account_id, risk_type, risk_level, description, action_taken
        ) VALUES (
            p_account_id, '日累计超限', 2,
            CONCAT('日累计交易 ', v_daily_total + p_amount, ' 超过限额 50000'),
            '拒绝交易'
        );
        RETURN FALSE;
    END IF;

    -- 检查异常交易模式
    IF EXISTS (
        SELECT 1 FROM transactions
        WHERE from_account_id = p_account_id
        AND created_at >= DATE_SUB(NOW(), INTERVAL 1 HOUR)
        GROUP BY DATE(created_at), HOUR(created_at)

            '人工审核'
        );
        -- 这里可以设置为需要人工审核，而不是直接拒绝
    END IF;

    RETURN TRUE;
```

```java
@Service
public class RiskControlService {

    @Autowired
    private JdbcTemplate jdbcTemplate;

    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    @Autowired
    private RiskRuleMapper riskRuleMapper;

    // 实时风控检测
    public RiskResult checkRisk(TransactionRequest request) {
        RiskResult result = new RiskResult();
        result.setTransactionId(request.getTransactionId());

        // 获取所有启用的风控规则
        List<RiskRule> rules = riskRuleMapper.selectEnabledRules();

        int totalRiskScore = 0;
        List<RiskRule> triggeredRules = new ArrayList<>();

        for (RiskRule rule : rules) {
            RiskRuleExecution execution = evaluateRule(rule, request);

            // 记录规则执行
            riskRuleMapper.insertExecution(execution);

            if (execution.getExecutionResult().equals("triggered")) {
                triggeredRules.add(rule);
                totalRiskScore += execution.getRiskScore();
            }
        }

        // 计算最终风控结果
        result.setRiskScore(totalRiskScore);
        result.setTriggeredRules(triggeredRules);

        // 根据风险分数和触发规则决定最终动作
        String action = determineAction(totalRiskScore, triggeredRules);
        result.setAction(action);

        // 记录风控日志
        if (!action.equals("allow")) {
            RiskControlLog log = new RiskControlLog();
            log.setCustomerId(request.getCustomerId());
            log.setAccountId(request.getAccountId());
            log.setTransactionId(request.getTransactionId());
            log.setRiskType("composite");
            log.setRiskLevel(totalRiskScore > 80 ? 3 : (totalRiskScore > 50 ? 2 : 1));
            log.setDescription("复合风控检测");
            log.setActionTaken(action);
            riskControlLogMapper.insert(log);
        }

        return result;
    }

    private RiskRuleExecution evaluateRule(RiskRule rule, TransactionRequest request) {
        RiskRuleExecution execution = new RiskRuleExecution();
        execution.setRuleId(rule.getId());
        execution.setCustomerId(request.getCustomerId());
        execution.setAccountId(request.getAccountId());
        execution.setTransactionId(request.getTransactionId());

        long startTime = System.currentTimeMillis();

        try {
            // 解析规则条件
            String condition = rule.getConditionExpression();

            // 根据规则类型执行不同的检测逻辑
            boolean triggered = false;
            int riskScore = 0;

            switch (rule.getRuleType()) {
                case "amount_limit":
                    triggered = checkAmountLimit(condition, request);
                    riskScore = triggered ? 30 : 0;
                    break;
                case "frequency":
                    triggered = checkFrequency(condition, request);
                    riskScore = triggered ? 40 : 0;
                    break;
                case "location":
                    triggered = checkLocation(condition, request);
                    riskScore = triggered ? 25 : 0;
                    break;
                case "device":
                    triggered = checkDevice(condition, request);
                    riskScore = triggered ? 20 : 0;
                    break;
            }

            execution.setExecutionResult(triggered ? "triggered" : "normal");
            execution.setRiskScore(riskScore);

        } catch (Exception e) {
            execution.setExecutionResult("error");
            execution.setRiskScore(0);
        }

        execution.setExecutionTime(System.currentTimeMillis() - startTime);
        return execution;
    }

    private boolean checkAmountLimit(String condition, TransactionRequest request) {
        // 解析条件，例如：amount > 10000
        // 这里简化处理，实际项目中需要更复杂的条件解析
        return request.getAmount() > 10000;
    }

    private boolean checkFrequency(String condition, TransactionRequest request) {
        // 检查交易频率
        String key = "risk:frequency:" + request.getCustomerId();
        Long count = redisTemplate.opsForValue().increment(key);

        if (count == 1) {
            redisTemplate.expire(key, 1, TimeUnit.HOURS);
        }

        return count > 10; // 1小时内超过10次交易
    }

    private boolean checkLocation(String condition, TransactionRequest request) {
        // 检查交易位置是否异常
        String key = "risk:location:" + request.getCustomerId();
        String lastLocation = (String) redisTemplate.opsForValue().get(key);

        if (lastLocation != null && !lastLocation.equals(request.getLocation())) {
            redisTemplate.opsForValue().set(key, request.getLocation(), 24, TimeUnit.HOURS);
            return true; // 位置异常
        }

        redisTemplate.opsForValue().set(key, request.getLocation(), 24, TimeUnit.HOURS);
        return false;
    }

    private boolean checkDevice(String condition, TransactionRequest request) {
        // 检查设备是否异常
        String key = "risk:device:" + request.getCustomerId();
        Set<String> knownDevices = (Set<String>) redisTemplate.opsForValue().get(key);

        if (knownDevices == null) {
            knownDevices = new HashSet<>();
            knownDevices.add(request.getDeviceId());
            redisTemplate.opsForValue().set(key, knownDevices, 30, TimeUnit.DAYS);
            return false;
        }

        if (!knownDevices.contains(request.getDeviceId())) {
            knownDevices.add(request.getDeviceId());
            redisTemplate.opsForValue().set(key, knownDevices, 30, TimeUnit.DAYS);
            return true; // 新设备，可能有风险
        }

        return false;
    }
}
```

**实时告警服务**：
```java
@Service
public class AlertService {

    @Autowired
    private JdbcTemplate jdbcTemplate;

    @Autowired
    private AlertRuleMapper alertRuleMapper;

    @KafkaListener(topics = "device-metrics")
    public void processDeviceMetrics(String message) {
        DeviceMetric metric = DeviceMetric.fromJson(message);

        // 获取该设备的告警规则
        List<AlertRule> rules = alertRuleMapper.selectByDeviceType(metric.getDeviceType());

        for (AlertRule rule : rules) {
            if (shouldTriggerAlert(rule, metric)) {
                createAlert(rule, metric);
            }
        }
    }

    private boolean shouldTriggerAlert(AlertRule rule, DeviceMetric metric) {
        // 检查规则条件
        switch (rule.getConditionType()) {
            case "threshold":
                return checkThresholdCondition(rule, metric);
            case "trend":
                return checkTrendCondition(rule, metric);
            case "anomaly":
                return checkAnomalyCondition(rule, metric);
            default:
                return false;
        }
    }

    private boolean checkThresholdCondition(AlertRule rule, DeviceMetric metric) {
        // 检查阈值条件
        double value = metric.getMetricValue().doubleValue();
        double threshold = rule.getThresholdValue();
        String operator = rule.getOperator();

        switch (operator) {
            case ">":
                return value > threshold;
            case "<":
                return value < threshold;
            case ">=":
                return value >= threshold;
            case "<=":
                return value <= threshold;
            case "==":
                return value == threshold;
            default:
                return false;
        }
    }

    private boolean checkTrendCondition(AlertRule rule, DeviceMetric metric) {
        // 检查趋势条件（连续上升或下降）
        String sql = "SELECT metric_value, timestamp FROM device_metrics " +
                    "WHERE device_id = ? AND metric_name = ? " +
                    "ORDER BY timestamp DESC LIMIT 5";

        List<Map<String, Object>> recentValues = jdbcTemplate.queryForList(
            sql, metric.getDeviceId(), metric.getMetricName());

        if (recentValues.size() < 3) {
            return false;
        }

        // 检查是否连续上升或下降
        boolean increasing = true;
        boolean decreasing = true;

        for (int i = 1; i < recentValues.size(); i++) {
            double prev = ((BigDecimal) recentValues.get(i-1).get("metric_value")).doubleValue();
            double curr = ((BigDecimal) recentValues.get(i).get("metric_value")).doubleValue();

            if (curr <= prev) increasing = false;
            if (curr >= prev) decreasing = false;
        }

        return (rule.getTrendType().equals("increasing") && increasing) ||
               (rule.getTrendType().equals("decreasing") && decreasing);
    }

    private boolean checkAnomalyCondition(AlertRule rule, DeviceMetric metric) {
        // 使用简单的统计方法检测异常
        String sql = "SELECT AVG(metric_value) as avg_value, STDDEV(metric_value) as std_dev " +
                    "FROM device_metrics " +
                    "WHERE device_id = ? AND metric_name = ? " +
                    "AND timestamp >= DATE_SUB(NOW(), INTERVAL 1 HOUR)";

        Map<String, Object> stats = jdbcTemplate.queryForMap(
            sql, metric.getDeviceId(), metric.getMetricName());

        double avg = ((BigDecimal) stats.get("avg_value")).doubleValue();
        double stdDev = ((BigDecimal) stats.get("std_dev")).doubleValue();
        double value = metric.getMetricValue().doubleValue();

        // 如果当前值偏离平均值超过 3 个标准差，认为是异常
        return Math.abs(value - avg) > 3 * stdDev;
    }

    private void createAlert(AlertRule rule, DeviceMetric metric) {
        // 检查是否已有未解决的相同告警
        String checkSql = "SELECT COUNT(*) FROM device_alerts " +
                         "WHERE device_id = ? AND alert_type = ? AND status = 0";

        Integer existingCount = jdbcTemplate.queryForObject(
            checkSql, Integer.class, metric.getDeviceId(), rule.getAlertType());

        if (existingCount != null && existingCount > 0) {
            return; // 已有未解决的告警，不重复创建
        }

        // 创建新告警
        String insertSql = "INSERT INTO device_alerts (device_id, alert_type, alert_level, " +
                         "alert_message, metric_name, metric_value, threshold_value, status) " +
                         "VALUES (?, ?, ?, ?, ?, ?, ?, 0)";

        jdbcTemplate.update(insertSql,
            metric.getDeviceId(),
            rule.getAlertType(),
            rule.getAlertLevel(),
            rule.getAlertMessage(),
            metric.getMetricName(),
            metric.getMetricValue(),
            rule.getThresholdValue()
        );

        // 发送告警通知
        sendAlertNotification(metric.getDeviceId(), rule.getAlertType(), rule.getAlertLevel());
    }
}
```

```sql
WHERE timestamp >= DATE_SUB(CURDATE(), INTERVAL 30 DAY)
AND device_id IN (
    SELECT device_id FROM devices
    WHERE device_type = 'pressure_sensor' AND status = 1
)
GROUP BY device_id, DATE(timestamp), metric_name
ORDER BY device_id, metric_date;

-- 预测性维护分析
SELECT /*+ TIFLASH() */
    d.device_id,
    d.device_name,
    d.device_type,
    d.location,
    d.last_seen,
    COUNT(DISTINCT CASE WHEN a.alert_type = 'temperature_high' THEN a.id END) as temp_alerts,
    COUNT(DISTINCT CASE WHEN a.alert_type = 'vibration_anomaly' THEN a.id END) as vibration_alerts,
    COUNT(DISTINCT CASE WHEN a.alert_type = 'pressure_low' THEN a.id END) as pressure_alerts,
    AVG(m.metric_value) as avg_temperature,
    MAX(m.metric_value) as max_temperature,
    CASE
        WHEN COUNT(DISTINCT CASE WHEN a.alert_type IN ('temperature_high', 'vibration_anomaly') THEN a.id END) > 10
        THEN '需要立即维护'
        WHEN COUNT(DISTINCT CASE WHEN a.alert_type IN ('temperature_high', 'vibration_anomaly') THEN a.id END) > 5
        THEN '建议近期维护'
        WHEN DATEDIFF(NOW(), MAX(mm.next_maintenance_date)) < 7
        THEN '即将到期维护'
        ELSE '运行正常'
    END as maintenance_priority
FROM devices d
LEFT JOIN device_alerts a ON d.device_id = a.device_id AND a.created_at >= DATE_SUB(CURDATE(), INTERVAL 30 DAY)
LEFT JOIN device_metrics m ON d.device_id = m.device_id
    AND m.metric_name = 'temperature'
    AND m.timestamp >= DATE_SUB(CURDATE(), INTERVAL 7 DAY)
LEFT JOIN device_maintenance mm ON d.device_id = mm.device_id AND mm.status IN (1, 2)
WHERE d.status = 1
GROUP BY d.device_id, d.device_name, d.device_type, d.location, d.last_seen
ORDER BY maintenance_priority DESC;
```

#### 4.1.3 实时排行榜系统

**排行榜更新服务**：
```java
@Service
public class LeaderboardService {

    @Autowired
    private JdbcTemplate jdbcTemplate;

    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    // 更新排行榜
    @Scheduled(fixedRate = 60000) // 每分钟更新一次
    public void updateLeaderboards() {
        updateLevelLeaderboard();
        updateCoinsLeaderboard();
        updateBattleWinsLeaderboard();
    }

    private void updateLevelLeaderboard() {
        // 获取等级前 1000 名玩家
        String sql = "SELECT player_id, level, experience " +
                    "FROM players " +
                    "WHERE status = 1 " +
                    "ORDER BY level DESC, experience DESC " +
                    "LIMIT 1000";

        List<Map<String, Object>> topPlayers = jdbcTemplate.queryForList(sql);

        // 清空旧数据
        jdbcTemplate.update("DELETE FROM leaderboards WHERE leaderboard_type = 'level'");

        // 批量插入新排名
        String insertSql = "INSERT INTO leaderboards (leaderboard_type, player_id, score, rank_position) " +
                          "VALUES ('level', ?, ?, ?)";

        jdbcTemplate.batchUpdate(insertSql, topPlayers, 100, (ps, player, i) -> {
            ps.setString(1, (String) player.get("player_id"));
            ps.setLong(2, ((Integer) player.get("level")).longValue());
            ps.setInt(3, i + 1);
        });

        // 更新 Redis 缓存
        updateRedisLeaderboard("level:ranking", topPlayers);
    }

    private void updateCoinsLeaderboard() {
        String sql = "SELECT player_id, coins " +
                    "FROM players " +
                    "WHERE status = 1 " +
                    "ORDER BY coins DESC " +
                    "LIMIT 1000";

        List<Map<String, Object>> topPlayers = jdbcTemplate.queryForList(sql);

        jdbcTemplate.update("DELETE FROM leaderboards WHERE leaderboard_type = 'coins'");

        String insertSql = "INSERT INTO leaderboards (leaderboard_type, player_id, score, rank_position) " +
                          "VALUES ('coins', ?, ?, ?)";

        jdbcTemplate.batchUpdate(insertSql, topPlayers, 100, (ps, player, i) -> {
            ps.setString(1, (String) player.get("player_id"));
            ps.setLong(2, ((Long) player.get("coins")));
            ps.setInt(3, i + 1);
        });

        updateRedisLeaderboard("coins:ranking", topPlayers);
    }

    private void updateRedisLeaderboard(String key, List<Map<String, Object>> players) {
        redisTemplate.delete(key);

        Map<String, Double> scores = new HashMap<>();
        for (Map<String, Object> player : players) {
            scores.put((String) player.get("player_id"), ((Number) player.get("score")).doubleValue());
        }

        redisTemplate.opsForZSet().add(key, scores);
        redisTemplate.expire(key, 1, TimeUnit.HOURS);
    }

    // 获取玩家排名
    public int getPlayerRank(String leaderboardType, String playerId) {
        // 先从 Redis 获取
        Long rank = redisTemplate.opsForZSet().reverseRank(leaderboardType + ":ranking", playerId);

        if (rank != null) {
            return rank.intValue() + 1;
        }

        // Redis 中没有，从数据库查询
        String sql = "SELECT rank_position FROM leaderboards " +
                    "WHERE leaderboard_type = ? AND player_id = ?";

        Integer dbRank = jdbcTemplate.queryForObject(sql, Integer.class, leaderboardType, playerId);

        return dbRank != null ? dbRank : -1;
    }

    // 获取排行榜前 N 名
    public List<LeaderboardEntry> getTopPlayers(String leaderboardType, int limit) {
        // 从 Redis 获取
        Set<ZSetOperations.TypedTuple<Object>> topSet = redisTemplate.opsForZSet()
            .reverseRangeWithScores(leaderboardType + ":ranking", 0, limit - 1);

        if (topSet != null && !topSet.isEmpty()) {
            return topSet.stream()
                .map(tuple -> new LeaderboardEntry(
                    (String) tuple.getValue(),
                    tuple.getScore().longValue(),
                    topSet.size() - new ArrayList<>(topSet).indexOf(tuple)
                ))
                .collect(Collectors.toList());
        }

        // 从数据库获取
        String sql = "SELECT l.player_id, l.score, l.rank_position, p.username " +
                    "FROM leaderboards l " +
                    "JOIN players p ON l.player_id = p.player_id " +
                    "WHERE l.leaderboard_type = ? " +
                    "ORDER BY l.rank_position " +
                    "LIMIT ?";

        return jdbcTemplate.query(sql, (rs, rowNum) -> {
            LeaderboardEntry entry = new LeaderboardEntry();
            entry.setPlayerId(rs.getString("player_id"));
            entry.setUsername(rs.getString("username"));
            entry.setScore(rs.getLong("score"));
            entry.setRank(rs.getInt("rank_position"));
            return entry;
        }, leaderboardType, limit);
    }
}
```
